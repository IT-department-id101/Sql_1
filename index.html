  • Salary: مووچە (int) – نرخی DEFAULT وەک ٣٠٠ دانراوە (ئەگەر نرخێک دانەنرێت، بە خۆکاری ٣٠٠ دەبێت)

📝 ڕوونکردنەوەی تەواو

12345678910
-- دروستکردنی خشتەی کارمەند
create table emp
(
  -- ژمارەی کارمەند
  empNo   int,
  -- ناوی کارمەند
  empName NVARCHAR(14),
  -- ناونیشانی کارمەند
  Address NVARCHAR(13),
  -- مووچە، بەهای بنەڕەتی 300
  Salary  int default 300
);

1️⃣ empNo – int – ژمارەی تەواو | نموونە: 1, 2, 100

2️⃣ empName – NVARCHAR(14) – NVARCHAR پشتگیری لە پیتە کوردی و عەرەبییەکان دەکات

3️⃣ Address – NVARCHAR(13) | نموونە: "سلێمانی"

4️⃣ Salary – DEFAULT 300 واتە ئەگەر بەهایەک نەنووسرا ٣٠٠ دەبێت

⚠️ خاڵی گرنگ:

  • هیچ سنوورێکی نییە – دوو کارمەند دوو هەمان ژمارەی هەبن ❌
  • بۆ باشترکردن سنوورەکان (Constraints) زیاد بکرێن

create table employees(
 employee_id    int,
 last_name      nvarchar(25) not null,
 email          nvarchar(25),
 salary         decimal(5,2),
 commission_pct decimal(2,2),
 hire_date      date not null,
 constraint emp_email_uk unique(email)
);
دەرەنجام: / Output:
زیادکرا
ڕوونکردنەوە:
  • خشتەیەکی نوێ بە ناوی employees بە چەندین سنوور دروست دەکات
  • employee_id : ژمارەی کارمەند (int) – هیچ سنوورێکی نییە لێرەدا
  • last_name : ناوی کۆتایی (nvarchar(25)) – not null واتە پێویستە، نابێت بەتاڵ بهێڵدرێتەوە
  • email : ئیمەیڵ (nvarchar(25)) – سنووری unique لەسەری دانراوە واتە هەر ئیمەیڵێک تەنها جارێک بەکاردێت
  • salary : مووچە (decimal(5,2)) – ٥ ژمارەی گشتی لەگەڵ ٢ ژمارەی دەیی، نموونە: 850.50
  • commission_pct : ڕێژەی کۆمیسیۆن (decimal(2,2)) – نموونە: 0.10 واتە ١٠٪
  • hire_date : بەرواری دامەزراندن (date) – not null واتە پێویستە
  • constraint emp_email_uk unique(email) : ئەمە سنووری تەبلەیە (table-level) – emp_email_uk ناوی سنوورەکەیە و دەتوانیت هەر ناوێکی تر بەکاربهێنیت

  • salary : مووچە (decimal(5,2)) – نموونە: 800.00
  • commission_pct : کۆمیسیۆن (decimal(2,2)) – نموونە: 0.10
  • hire_date : بەرواری دامەزراندن (date) – not null پێویستە
  • Age : تەمەن (int) – سنووری check لەسەری دانراوە
  • constraint age_chk_cons check (Age between 0 and 150) : واتە تەمەن دەبێت لە نێوان ٠ و ١٥٠ بێت – ئەگەر -٥ یان ٢٠٠ بنووسیت هەڵە دەدات

create table test
( empno       int          primary key,
  empname     varchar(20) not null,
  address     varchar(20),
  email       varchar(20) unique,
  city        varchar(20),
  certificate varchar(20),
  salary      int          default 300,
  age         int,
  constraint age_chk_cons check (age between 0 and 150)
);
دەرەنجام: / Output:
زیادکرا
ڕوونکردنەوە:
  • خشتەیەکی تەواو بە ناوی test دروست دەکات کە هەموو جۆرە سنوورەکانی تێدایە
  • empno : ژمارەی کارمەند (int) – primary key واتە بێهاوتا و پێویستە، نموونە: 1, 2, 3
  • empname : ناوی کارمەند (varchar(20)) – not null واتە پێویستە، نموونە: 'ئەحمەد'
  • address : ناونیشان (varchar(20)) – دڵخوازە، دەتوانیت بەتاڵ بهێڵدرێتەوە
  • email : ئیمەیڵ (varchar(20)) – unique واتە هەر ئیمەیڵێک تەنها بۆ یەک کارمەند بەکاردێت
  • city : شار (varchar(20)) – دڵخوازە، نموونە: 'سلێمانی'
  • certificate : بڕوانامە (varchar(20)) – دڵخوازە، نموونە: 'بەکالۆریۆس'
  • salary : مووچە (int) – default 300 واتە ئەگەر نرخی مووچە نەنووسرێت بە خۆکاری ٣٠٠ دەبێت
  • age : تەمەن (int) – check (age between 0 and 150) واتە تەنها نرخی ٠ تا ١٥٠ قبووڵ دەکرێت
